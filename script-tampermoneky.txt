// ==UserScript==
// @name         Google Form Auto Spam
// @namespace    http://tampermonkey.net/
// @version      14.0
// @description  Tự động random đáp án Trắc nghiệm và Text, kết hợp né mìn hoàn hảo
// @match        *://docs.google.com/forms/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    function nukeClick(element) {
        if (!element) return;

        element.scrollIntoView({ behavior: 'smooth', block: 'center' });

        setTimeout(() => {
            element.focus();
            const events = [
                new MouseEvent('mouseenter', { bubbles: true, cancelable: true, view: window }),
                new PointerEvent('pointerdown', { bubbles: true, cancelable: true }),
                new MouseEvent('mousedown', { bubbles: true, cancelable: true, buttons: 1, view: window }),
                new PointerEvent('pointerup', { bubbles: true, cancelable: true }),
                new MouseEvent('mouseup', { bubbles: true, cancelable: true, buttons: 1, view: window }),
                new MouseEvent('click', { bubbles: true, cancelable: true, view: window }),
                new KeyboardEvent('keydown', { key: 'Enter', bubbles: true, cancelable: true }),
                new KeyboardEvent('keyup', { key: 'Enter', bubbles: true, cancelable: true })
            ];

            events.forEach(ev => element.dispatchEvent(ev));
        }, 200);
    }

    setInterval(() => {
        let isFillingRightNow = false;

        // 1. Bấm nút "Gửi phản hồi khác"
        let links = document.querySelectorAll('a');
        for (let a of links) {
            let txt = a.textContent.toLowerCase();
            if (txt.includes('khác') || txt.includes('another')) {
                nukeClick(a);
                return;
            }
        }

        // 2. Điền Text NGẪU NHIÊN
        // Bạn có thể sửa các chữ trong ngoặc kép này theo ý muốn
        const randomTexts = ["Hoàn toàn đồng ý", "Đồng ý", "Bình thường", "Rất tốt", "Không có ý kiến", "Tuyệt vời", "Khá ổn"];

        document.querySelectorAll('input[type="text"], textarea').forEach(t => {
            if (t.offsetParent !== null && !t.value) {
                // Bốc ngẫu nhiên 1 câu trong mảng randomTexts
                let pickedText = randomTexts[Math.floor(Math.random() * randomTexts.length)];
                t.value = pickedText;
                t.dispatchEvent(new Event('input', { bubbles: true }));
                isFillingRightNow = true;
            }
        });

        // 3. Click Radio NGẪU NHIÊN (Trắc nghiệm 1 lựa chọn)
        document.querySelectorAll('div[role="radiogroup"]').forEach(group => {
            if (group.offsetParent !== null && !group.querySelector('div[aria-checked="true"]')) {
                let radios = group.querySelectorAll('div[role="radio"]');
                if (radios.length > 0) {
                    // Tung xúc xắc chọn 1 vị trí bất kỳ
                    let randomRadio = radios[Math.floor(Math.random() * radios.length)];
                    nukeClick(randomRadio);
                    isFillingRightNow = true;
                }
            }
        });

        // 4. Click Checkbox NGẪU NHIÊN (Trắc nghiệm nhiều lựa chọn)
        document.querySelectorAll('div[role="listitem"]').forEach(item => {
            if (item.offsetParent !== null) {
                let checkboxes = item.querySelectorAll('div[role="checkbox"]');
                if (checkboxes.length > 0 && !item.querySelector('div[aria-checked="true"]')) {
                    // Để nhanh gọn, bot bốc ngẫu nhiên 1 ô checkbox để tick
                    let randomCheckbox = checkboxes[Math.floor(Math.random() * checkboxes.length)];
                    nukeClick(randomCheckbox);
                    isFillingRightNow = true;
                }
            }
        });

        if (isFillingRightNow) return;

        // ==========================================
        // 5. TÌM NÚT (LƯỚI LỌC TỬ THẦN NÉ QUAY LẠI VÀ XÓA FORM)
        // ==========================================
        let targetBtn = null;

        let buttons = document.querySelectorAll('div[role="button"]');
        for (let b of buttons) {
            if (b.offsetParent !== null && b.getAttribute('aria-disabled') !== 'true') {
                let txt = b.innerText.trim().toLowerCase();

                if (txt.includes('quay lại') || txt.includes('back') || txt.includes('xóa') || txt.includes('clear')) {
                    continue;
                }

                if (txt === 'tiếp' || txt === 'next' || txt === 'gửi' || txt === 'submit') {
                    targetBtn = b;
                    break;
                }
            }
        }

        if (targetBtn) {
            console.log("🔥 [BOT] Đã chốt đúng nút GỬI/TIẾP bằng Text!");
            nukeClick(targetBtn);

            let coreSpan = targetBtn.querySelector('span.NPEfkd');
            if (coreSpan) nukeClick(coreSpan);

        } else {
            let actionButtons = Array.from(document.querySelectorAll('div[role="button"][jsaction*="click:cOuCgd"], div[role="button"][jscontroller="VXdfxd"]'));
            let visibleBtns = actionButtons.filter(b => b.offsetParent !== null && b.getAttribute('aria-disabled') !== 'true');

            if (visibleBtns.length > 0) {
                let realSubmitBtn = null;

                if (visibleBtns.length >= 3) {
                    realSubmitBtn = visibleBtns[1];
                } else if (visibleBtns.length === 2) {
                    realSubmitBtn = visibleBtns[0];
                } else {
                    realSubmitBtn = visibleBtns[0];
                }

                if (realSubmitBtn) {
                    console.log("🎯 [BOT] JSACTION: Đã khóa mục tiêu an toàn!");
                    nukeClick(realSubmitBtn);
                }
            }
        }

    }, 500);
})();
